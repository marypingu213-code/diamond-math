    <script>
        const canvas = document.getElementById('gridCanvas');
        const ctx = canvas.getContext('2d');

        // 畫布與網格設定
        const GRID_SIZE = 20;
        let canvasSize = 0;
        let cellSize = 0;
        let padding = 0;

        // 應用程式狀態
        const state = {
            center: null,       
            radius: 0,          
            isDragging: false,
            dragPixelPos: null, 
            phase: 'idle',      // 'idle', 'dragging', 'waiting', 'drawing', 'quiz'
            timeLeft: 5,        
            startAngle: 0,      
            drawAngle: 0        
        };

        let timerId = null;
        let countdownInterval = null;
        let animationFrameId = null;

        // 初始化與響應式調整大小
        function resizeCanvas() {
            // 判斷是否為寬螢幕 (桌機/平板橫式)
            const isDesktop = window.innerWidth >= 1024;
            // 如果是寬螢幕，要扣掉右邊測驗題面板的寬度(約400px)；手機版則滿版
            const availableWidth = isDesktop ? (window.innerWidth - 450) : (window.innerWidth - 32);
            
            // 限制最大為 550px，確保不管怎樣都不會過大
            const maxSize = Math.min(availableWidth, window.innerHeight - 200, 550);
            
            canvas.width = maxSize;
            canvas.height = maxSize;
            
            // 計算每一格的大小，保留外圍一格的 padding 讓邊界點可以完整顯示
            cellSize = maxSize / (GRID_SIZE + 2);
            padding = cellSize;
            
            draw();
        }

        window.addEventListener('resize', resizeCanvas);

        // 取得滑鼠或觸控在畫布上的相對座標
        function getPointerPos(e) {
            const rect = canvas.getBoundingClientRect();
            let clientX = e.clientX;
            let clientY = e.clientY;

            if (e.touches && e.touches.length > 0) {
                clientX = e.touches[0].clientX;
                clientY = e.touches[0].clientY;
            } else if (e.changedTouches && e.changedTouches.length > 0) {
                clientX = e.changedTouches[0].clientX;
                clientY = e.changedTouches[0].clientY;
            }

            return {
                x: clientX - rect.left,
                y: clientY - rect.top
            };
        }

        function getNearestGridPoint(pos) {
            let gridX = Math.round((pos.x - padding) / cellSize);
            let gridY = Math.round((pos.y - padding) / cellSize);
            gridX = Math.max(0, Math.min(GRID_SIZE, gridX));
            gridY = Math.max(0, Math.min(GRID_SIZE, gridY));
            return { x: gridX, y: gridY };
        }

        function startAction(e) {
            if (state.phase === 'quiz') return; // 測驗狀態時鎖定畫布
            if (e.type === 'touchstart') e.preventDefault(); 
            
            clearTimeout(timerId);
            clearInterval(countdownInterval);
            cancelAnimationFrame(animationFrameId);
            
            const pos = getPointerPos(e);
            state.center = getNearestGridPoint(pos);
            state.isDragging = true;
            state.phase = 'dragging';
            state.radius = 0;
            state.dragPixelPos = pos;
            
            draw();
        }

        function moveAction(e) {
            if (!state.isDragging) return;
            if (e.type === 'touchmove') e.preventDefault();
            
            const pos = getPointerPos(e);
            state.dragPixelPos = pos;
            
            const cx = padding + state.center.x * cellSize;
            const cy = padding + state.center.y * cellSize;
            const dx = pos.x - cx;
            const dy = pos.y - cy;
            const distanceInPixels = Math.sqrt(dx * dx + dy * dy);
            
            state.radius = Math.round(distanceInPixels / cellSize);
            draw();
        }

        function endAction(e) {
            if (!state.isDragging) return;
            state.isDragging = false;
            
            if (state.radius > 0) {
                const cx = padding + state.center.x * cellSize;
                const cy = padding + state.center.y * cellSize;
                const dx = state.dragPixelPos.x - cx;
                const dy = state.dragPixelPos.y - cy;
                const angle = Math.atan2(dy, dx);

                state.phase = 'waiting';
                state.timeLeft = 5;
                state.startAngle = angle;
                state.drawAngle = angle;
                
                draw();
                
                countdownInterval = setInterval(() => {
                    state.timeLeft -= 1;
                    draw();
                }, 1000);

                timerId = setTimeout(() => {
                    clearInterval(countdownInterval);
                    state.phase = 'drawing';
                    animateCircle();
                }, 5000);
            } else {
                state.phase = 'idle';
                state.center = null;
                draw();
            }
        }

        function animateCircle() {
            if (state.phase !== 'drawing') return;
            
            state.drawAngle += 0.08; 
            const endAngle = state.startAngle + (2 * Math.PI);
            
            if (state.drawAngle >= endAngle) {
                state.drawAngle = endAngle;
                state.phase = 'quiz';
                draw();
                showQuiz(); // 動畫畫完後呼叫測驗解鎖
                return;
            }
            
            draw();
            animationFrameId = requestAnimationFrame(animateCircle);
        }

        function draw() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // 游標樣式切換：測驗時顯示禁止符號
            canvas.style.cursor = state.phase === 'quiz' ? 'not-allowed' : 'crosshair';

            // 1. 繪製 20x20 方格紙
            ctx.beginPath();
            ctx.strokeStyle = '#bae6fd'; 
            ctx.lineWidth = 1;
            
            for (let i = 0; i <= GRID_SIZE; i++) {
                const pos = padding + i * cellSize;
                ctx.moveTo(pos, padding);
                ctx.lineTo(pos, padding + GRID_SIZE * cellSize);
                ctx.moveTo(padding, pos);
                ctx.lineTo(padding + GRID_SIZE * cellSize, pos);
            }
            ctx.stroke();

            if (!state.center) return;

            const cx = padding + state.center.x * cellSize;
            const cy = padding + state.center.y * cellSize;

            // 2. 繪製圓心 (黑點)
            ctx.beginPath();
            ctx.fillStyle = '#111827';
            ctx.arc(cx, cy, 6, 0, Math.PI * 2);
            ctx.fill();

            // 3. 狀態繪製
            if (state.phase === 'dragging') {
                ctx.beginPath();
                ctx.strokeStyle = '#9ca3af';
                ctx.setLineDash([5, 5]);
                ctx.moveTo(cx, cy);
                if (state.dragPixelPos) {
                    ctx.lineTo(state.dragPixelPos.x, state.dragPixelPos.y);
                }
                ctx.stroke();
                ctx.setLineDash([]);

                if (state.radius > 0) {
                    // 游標旁邊顯示目前半徑
                    ctx.fillStyle = '#4b5563'; 
                    ctx.font = 'bold 16px sans-serif';
                    ctx.textAlign = 'left';
                    ctx.textBaseline = 'bottom';
                    
                    // 加一個白色半透明背景讓字更清楚
                    const text = `半徑: ${state.radius}`;
                    const textWidth = ctx.measureText(text).width;
                    ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
                    ctx.fillRect(state.dragPixelPos.x + 5, state.dragPixelPos.y - 30, textWidth + 10, 24);
                    
                    ctx.fillStyle = '#1e3a8a';
                    ctx.fillText(text, state.dragPixelPos.x + 10, state.dragPixelPos.y - 10);

                    ctx.beginPath();
                    ctx.strokeStyle = 'rgba(59, 130, 246, 0.4)';
                    ctx.lineWidth = 2;
                    ctx.arc(cx, cy, state.radius * cellSize, 0, Math.PI * 2);
                    ctx.stroke();
                }
            } 
            else if (state.phase === 'waiting') {
                ctx.beginPath();
                ctx.strokeStyle = '#9ca3af';
                ctx.setLineDash([5, 5]);
                ctx.moveTo(cx, cy);
                ctx.lineTo(cx + Math.cos(state.startAngle) * state.radius * cellSize, cy + Math.sin(state.startAngle) * state.radius * cellSize);
                ctx.stroke();

                ctx.beginPath();
                ctx.strokeStyle = 'rgba(59, 130, 246, 0.2)'; 
                ctx.setLineDash([4, 4]);
                ctx.lineWidth = 2;
                ctx.arc(cx, cy, state.radius * cellSize, 0, Math.PI * 2);
                ctx.stroke();
                ctx.setLineDash([]);

                ctx.fillStyle = '#ef4444'; 
                ctx.font = 'bold 28px sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillText(state.timeLeft, cx, cy - 25);
            } 
            else if (state.phase === 'drawing' || state.phase === 'quiz') {
                // 畫出半徑起點的虛線 (讓學生可以數格子)
                ctx.beginPath();
                ctx.strokeStyle = '#9ca3af';
                ctx.setLineDash([5, 5]);
                ctx.moveTo(cx, cy);
                ctx.lineTo(cx + Math.cos(state.startAngle) * state.radius * cellSize, cy + Math.sin(state.startAngle) * state.radius * cellSize);
                ctx.stroke();
                ctx.setLineDash([]);

                ctx.beginPath();
                ctx.strokeStyle = '#3b82f6'; 
                ctx.lineWidth = 3;
                ctx.arc(cx, cy, state.radius * cellSize, state.startAngle, state.drawAngle);
                ctx.stroke();

                if (state.phase === 'drawing') {
                    const px = cx + Math.cos(state.drawAngle) * state.radius * cellSize;
                    const py = cy + Math.sin(state.drawAngle) * state.radius * cellSize;
                    drawCompass(cx, cy, px, py, state.radius * cellSize);
                }
            }
        }

        // 繪製虛化圓規
        function drawCompass(cx, cy, px, py, radiusPx) {
            ctx.save();
            ctx.globalAlpha = 0.6; 
            const compassHeight = Math.max(radiusPx * 1.2, 80);
            const hx = cx;
            const hy = cy - compassHeight;

            // 針腳
            ctx.beginPath();
            ctx.moveTo(cx, cy);
            ctx.lineTo(hx, hy);
            ctx.strokeStyle = '#9ca3af'; 
            ctx.lineWidth = 4;
            ctx.lineCap = 'round';
            ctx.stroke();

            // 筆腳
            ctx.beginPath();
            ctx.moveTo(px, py);
            ctx.lineTo(hx, hy);
            ctx.strokeStyle = '#4b5563'; 
            ctx.lineWidth = 5;
            ctx.lineCap = 'round';
            ctx.stroke();

            // 鉛筆頭
            const dxLeg = px - hx;
            const dyLeg = py - hy;
            const legLen = Math.sqrt(dxLeg * dxLeg + dyLeg * dyLeg);
            const nx = dxLeg / legLen;
            const ny = dyLeg / legLen;
            const baseDist = 18; 
            const bx = px - nx * baseDist;
            const by = py - ny * baseDist;
            const perpX = -ny;
            const perpY = nx;
            const halfWidth = 5;

            ctx.beginPath();
            ctx.moveTo(px, py);
            ctx.lineTo(bx + perpX * halfWidth, by + perpY * halfWidth);
            ctx.lineTo(bx - perpX * halfWidth, by - perpY * halfWidth);
            ctx.closePath();
            ctx.fillStyle = '#fcd34d'; 
            ctx.fill();
            
            ctx.beginPath();
            ctx.moveTo(px, py);
            ctx.lineTo(px - nx * 5 + perpX * 1.5, py - ny * 5 + perpY * 1.5);
            ctx.lineTo(px - nx * 5 - perpX * 1.5, py - ny * 5 - perpY * 1.5);
            ctx.closePath();
            ctx.fillStyle = '#111827';
            ctx.fill();

            // 頂部轉軸
            ctx.beginPath();
            ctx.arc(hx, hy, 7, 0, Math.PI * 2);
            ctx.fillStyle = '#374151'; 
            ctx.fill();
            ctx.strokeStyle = '#d1d5db';
            ctx.lineWidth = 2;
            ctx.stroke();

            // 頂部把手
            ctx.beginPath();
            ctx.moveTo(hx, hy - 7);
            ctx.lineTo(hx, hy - 28);
            ctx.strokeStyle = '#1f2937';
            ctx.lineWidth = 5;
            ctx.lineCap = 'round';
            ctx.stroke();

            ctx.beginPath();
            ctx.arc(hx, hy - 28, 5, 0, Math.PI * 2);
            ctx.fillStyle = '#1f2937';
            ctx.fill();
            ctx.restore();
        }

        // --- 測驗題相關邏輯 ---
        function showQuiz() {
            // 隱藏未解鎖的遮罩
            document.getElementById('quizDisabledOverlay').classList.add('opacity-0');
            setTimeout(() => {
                document.getElementById('quizDisabledOverlay').classList.add('hidden');
            }, 500); // 等待淡出動畫結束

            // 改變面板樣式以吸引注意
            const panel = document.getElementById('quizPanel');
            panel.classList.replace('border-gray-200', 'border-blue-400');
            panel.classList.add('ring-4', 'ring-blue-100');

            // 重置輸入與回饋
            document.getElementById('ansRadius').value = '0';
            document.getElementById('ansDiameter').value = '0';
            document.getElementById('quizFeedback').innerText = '';
            document.getElementById('mathConcept').classList.add('hidden'); // 重置時隱藏概念區塊
            document.getElementById('checkBtn').classList.remove('hidden');
            document.getElementById('resetBtn').classList.add('hidden');

            // 如果是手機/平板直向 (螢幕寬度小於 1024px)，自動捲動畫面到測驗題的位置
            if (window.innerWidth < 1024) {
                setTimeout(() => {
                    panel.scrollIntoView({ behavior: 'smooth', block: 'center' });
                }, 100);
            }
        }

        function changeValue(id, delta) {
            const input = document.getElementById(id);
            let val = parseInt(input.value) || 0;
            val += delta;
            
            // 確保數字不會小於 0
            if (val < 0) val = 0; 
            input.value = val;
            
            // 學生每次更改數字時，自動清空下方的錯誤提示文字與概念區塊，提升體驗
            document.getElementById('quizFeedback').innerText = '';
            document.getElementById('mathConcept').classList.add('hidden');
        }

        function checkAnswers() {
            const rInput = parseInt(document.getElementById('ansRadius').value);
            const dInput = parseInt(document.getElementById('ansDiameter').value);
            const feedback = document.getElementById('quizFeedback');

            if (isNaN(rInput) || isNaN(dInput)) {
                feedback.innerText = '⚠️ 請在兩個框框都輸入數字喔！';
                feedback.className = 'h-6 text-sm font-bold text-orange-500 text-center w-full';
                document.getElementById('mathConcept').classList.add('hidden');
                return;
            }

            const correctR = state.radius;
            const correctD = state.radius * 2;

            if (rInput === correctR && dInput === correctD) {
                feedback.innerText = '🎉 答對了！太棒了！ 🎉';
                feedback.className = 'h-6 text-lg font-bold text-green-600 text-center w-full';
                document.getElementById('mathConcept').classList.remove('hidden'); // 答對時顯示概念區塊
                document.getElementById('checkBtn').classList.add('hidden');
                document.getElementById('resetBtn').classList.remove('hidden');
            } else if (rInput === correctR && dInput !== correctD) {
                feedback.innerText = '🤔 半徑對了，但直徑算錯囉！再想想！';
                feedback.className = 'h-6 text-sm font-bold text-red-500 text-center w-full';
                document.getElementById('mathConcept').classList.add('hidden');
            } else if (rInput !== correctR && dInput === correctD) {
                feedback.innerText = '🤔 直徑對了，但半徑寫錯囉！再檢查一下左邊圖形！';
                feedback.className = 'h-6 text-sm font-bold text-red-500 text-center w-full';
                document.getElementById('mathConcept').classList.add('hidden');
            } else {
                feedback.innerText = '❌ 兩個答案都不對喔，數一下左邊藍色虛線有幾格！';
                feedback.className = 'h-6 text-sm font-bold text-red-500 text-center w-full';
                document.getElementById('mathConcept').classList.add('hidden');
            }
        }
